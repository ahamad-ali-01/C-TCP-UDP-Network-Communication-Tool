cmake_minimum_required(VERSION 3.16)
project(NetPulse VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NETPULSE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
add_executable(netpulse src/main.cpp src/protocol.cpp src/statistics.cpp src/net.cpp)
target_include_directories(netpulse PRIVATE include)
target_compile_options(netpulse PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(netpulse PRIVATE pthread)
if(NETPULSE_SANITIZERS)
  target_compile_options(netpulse PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(netpulse PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(netpulse_tests tests/test_main.cpp src/protocol.cpp src/statistics.cpp)
target_include_directories(netpulse_tests PRIVATE include)
target_compile_options(netpulse_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME unit_tests COMMAND netpulse_tests)
add_test(NAME integration_tests COMMAND bash ${CMAKE_SOURCE_DIR}/scripts/integration_test.sh $<TARGET_FILE:netpulse>)
set_tests_properties(integration_tests PROPERTIES TIMEOUT 30)
install(TARGETS netpulse RUNTIME DESTINATION bin)
